    double startTime = CycleTimer::currentSeconds();
    int height = static_cast<int>(args->height);

    // блочное разбиение: каждый поток берёт подряд идущие строки
    int rowsPerThread = (height + args->numThreads - 1) / args->numThreads;
    int startRow = args->threadId * rowsPerThread;
    int numRows = std::min(rowsPerThread, height - startRow);
    if (numRows > 0) {
        mandelbrotSerial(args->x0, args->y0, args->x1, args->y1,
                         args->width, args->height,
                         startRow, numRows, args->maxIterations, args->output);
    }

    double endTime = CycleTimer::currentSeconds();
    printf("Thread %d: %.3f ms\n", args->threadId, (endTime - startTime) * 1000);
